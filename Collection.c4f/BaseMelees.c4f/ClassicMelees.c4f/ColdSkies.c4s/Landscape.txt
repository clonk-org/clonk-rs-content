/* Dynamische Himmelsinseln */

// Basic island shape
overlay Islands { grp=1; invert=1;
  // islands
  overlay { algo=bozo; a=10; invert=1; zoomX=50; zoomY=-10; turbulence=100000; } &
  overlay { algo=rndchecker; a=1; rotate=125; zoomX=-20; zoomY=-20; turbulence=100; } &
  // loose bounds
  overlay { x=5; y= 15; wdt=90; hgt=85; loosebounds=1; turbulence=10; };
  };

// Basic InEarth
overlay InEarth { algo=bozo; a=8; turbulence=10000; };

// randomly alter mat
overlay AlterMat { algo=rndchecker; zoomX=-100; zoomY=-100; };


// skylands map
map SkyLands {


  // Group 1: base=rock
  Islands & overlay { mat=Rock; tex=Rough;

    // ice top-layers
    overlay { algo=border; a=0; b=1; turbulence=100; mat=Ice; tex=Smooth; oy=-1; };

    // coal
    overlay { algo=border; invert=1; a=1; b=1; turbulence=1000; mat=Coal; tex=Smooth; };

    // gold
    InEarth { mat=Gold; tex=Rough; };

    // ore
    InEarth { mat=Ore; tex=Ridge; AlterMat { mat=Ore; tex=Rough; }; };

    // overlay ice
    AlterMat & overlay { algo=border; a=3; b=3; turbulence=10; mat=Ice; tex=Smooth; };
    };


  // Group 2: base=ice
  Islands & overlay { mat=Snow; tex=Rough;

    // rock top-layers
    overlay { algo=border; a=0; b=1; turbulence=100; mat=Ice; tex=Smooth; oy=1;
      // inner core: Ice
      AlterMat { mat=Crystal; tex=Rough; };
      };

    // coal
    InEarth { mat=Coal; tex=Swirl; };

    // crystal
    InEarth { mat=Crystal; tex=Structure; a=5; sub=0; };

    // sulphur
    InEarth { mat=Sulphur; tex=Swirl; a=5; };

    // snow
    InEarth { mat=Snow; tex=Swirl; a=10;  };

    // overlay ice
    AlterMat & overlay { algo=border; a=1; b=1; turbulence=100; mat=Ice; tex=Smooth; } ;

    };


  };
