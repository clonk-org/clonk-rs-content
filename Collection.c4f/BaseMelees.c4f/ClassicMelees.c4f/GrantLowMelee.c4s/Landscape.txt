/* CivilHigh */

// Basic island shape
overlay Islands { grp=1; invert=1;
  // islands
			overlay { algo=bozo; a=4; turbulence=1000; loosebounds=1;
				overlay { algo=rndchecker; zoomX=-50; zoomY=-50; } |
				overlay { algo=random; a=5; };
				} &
  // loose bounds
  overlay { x=5; y=10; wdt=90; hgt=90; loosebounds=1; turbulence=10; };
  };

// Basic InEarth
overlay InEarth { algo=bozo; a=8; turbulence=10000; };

// randomly alter mat
overlay AlterMat { algo=rndchecker; zoomX=-50; zoomY=-50; };


// skylands map
map SkyLands {

  mat=Tunnel;
  overlay { mat=DuroLava; y=90; hgt=15; };

  // Group 1: base=rock
  Islands & overlay { mat=Rock; tex=Rough;

    // earth top-layers
    overlay { algo=border; a=1; b=1; turbulence=100; mat=Earth; tex=Smooth; oy=-1; };

    // Coal
    InEarth { mat=Coal; tex=Rough; };

    // Oil
    overlay { algo=border; invert=1; a=1; b=1; turbulence=100; mat=Oil; tex=Smooth; };

    // bounds for Oil
    overlay { algo=border; a=1; b=1; oy=1; mat=Earth; tex=Rough; };

    // ore
    InEarth { mat=Ore; tex=Ridge; AlterMat { mat=Ore; tex=Rough; }; };

    // overlay earth
    AlterMat & overlay { algo=border; a=3; b=3; turbulence=10; mat=Earth; tex=Smooth; };
    };


  // Group 2: base=earth
  Islands & overlay { mat=Earth; tex=Rough;

    // rock top-layers
    overlay { algo=border; a=0; b=1; turbulence=100; mat=Earth; tex=Smooth; oy=1;
      // inner rock: granite
      AlterMat { mat=Granite; tex=Rough; };
      };

    // Lava
    overlay { algo=border; invert=1; a=1; b=1; turbulence=100; mat=DuroLava; tex=Smooth; };

    // bounds for Lava
    overlay { algo=border; a=1; b=1; oy=1; mat=FlyAshes; tex=Rough; };

    // coal
    InEarth { mat=Coal; tex=Swirl; };

    // sulphur
    InEarth { mat=Sulphur; tex=Swirl; };

    // Coal
    InEarth { mat=Coal; tex=Structure; a=9; };

    // overlay earth
    AlterMat & overlay { algo=border; a=1; b=1; turbulence=100; mat=Earth; tex=Smooth; } ;

    };


  };
